import { NextResponse } from "next/server";
import prisma from "../../../lib/db";
import { Resend } from "resend";

const resend = new Resend(process.env.RESEND_API_KEY);

export async function POST(request: Request) {
  try {
    const body = await request.json();
    
    const { 
      firstName, lastName, customerEmail, customerPhone, 
      address1, address2, city, state, zip, country, 
      totalAmount, items 
    } = body;

    // 1. Save order to database
    const newOrder = await prisma.order.create({
      data: {
        firstName, lastName, customerEmail, customerPhone,
        address1, address2: address2 || null, city, state, zip, country,
        totalAmount: parseFloat(totalAmount),
        items: {
          create: items.map((item: any) => ({
            productId: item.productId || item.id,
            quantity: item.quantity,
            price: parseFloat(item.price)
          }))
        }
      }
    });

    // 2. SEND PROFESSIONAL RECEIPT TO THE CLIENT
    await resend.emails.send({
      from: "orders@authenticculturalearth.com", // Apna verified domain yahan likhein
      to: customerEmail, // Yeh dynamic email hai jo client form mein fill karega
      subject: `Order Confirmation - Authentic Cultural Earth (#${newOrder.id})`,
      html: `
        <div style="font-family: sans-serif; max-w: 600px; margin: 0 auto; color: #050505;">
          <h2 style="text-align: center; font-weight: normal; letter-spacing: 1px;">AUTHENTIC CULTURAL EARTH</h2>
          <hr style="border: 0; border-top: 1px solid #eaeaea; margin: 20px 0;" />
          
          <h3 style="font-size: 24px;">Thank you for your acquisition, ${firstName}.</h3>
          <p style="color: #666; line-height: 1.6;">We have successfully received your order. Your cultural packs are currently being prepared for dispatch.</p>
          
          <div style="background-color: #fbfaf8; padding: 20px; border-radius: 8px; margin: 20px 0;">
            <p style="margin: 0 0 10px 0; font-size: 12px; text-transform: uppercase; letter-spacing: 2px; color: #888;">Order Identifier</p>
            <p style="margin: 0; font-size: 16px; font-weight: bold;">${newOrder.id}</p>
          </div>

          <h4 style="text-transform: uppercase; letter-spacing: 1px; border-bottom: 1px solid #eaeaea; padding-bottom: 10px;">Order Summary</h4>
          <p><strong>Total Value:</strong> $${totalAmount} USD</p>
          
          <h4 style="text-transform: uppercase; letter-spacing: 1px; border-bottom: 1px solid #eaeaea; padding-bottom: 10px;">Shipping Destination</h4>
          <p style="color: #444; line-height: 1.6;">
            ${firstName} ${lastName}<br/>
            ${address1} ${address2 ? '<br/>' + address2 : ''}<br/>
            ${city}, ${state} ${zip}<br/>
            ${country}
          </p>

          <h4 style="text-transform: uppercase; letter-spacing: 1px; border-bottom: 1px solid #eaeaea; padding-bottom: 10px;">Delivery Estimate</h4>
          <p style="color: #444; line-height: 1.6;">
            Domestic orders typically arrive within 3-5 business days, while international shipping can take between 10-14 business days depending on the destination and customs processing.
          </p>
          
          <hr style="border: 0; border-top: 1px solid #eaeaea; margin: 30px 0;" />
          <p style="font-size: 12px; color: #888; text-align: center;">If you have any questions, please contact hello@authenticculturalearth.com</p>
        </div>
      `
    });

    // 3. SEND ALERT TO THE OWNER
    await resend.emails.send({
      from: "orders@authenticculturalearth.com", // Apna verified domain yahan likhein
      to: "authenticculturalearth@tuta.io", // Aapka apna inbox
      subject: `🎉 New Order Received! - $${totalAmount}`,
      html: `
        <h2>You have a new order!</h2>
        <p><strong>Order ID:</strong> ${newOrder.id}</p>
        <p><strong>Customer:</strong> ${firstName} ${lastName}</p>
        <p><strong>Email:</strong> ${customerEmail}</p>
        <p><strong>Total Value:</strong> $${totalAmount} USD</p>
        <h3>Shipping Details:</h3>
        <p>
          ${address1}<br/>
          ${city}, ${state} ${zip}<br/>
          ${country}
        </p>
      `
    });

    return NextResponse.json({ success: true, orderId: newOrder.id });
    
  } catch (error) {
    console.error("Order processing error:", error);
    return NextResponse.json({ error: "Checkout failed" }, { status: 500 });
  }
}
